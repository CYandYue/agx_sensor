cmake_minimum_required(VERSION 3.0.2)
project(lidar_rgbd_capture)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs geometry_msgs nav_msgs std_msgs livox_ros_driver2 tf2_ros tf2_eigen)
find_package(OpenCV REQUIRED COMPONENTS core calib3d)
find_package(Eigen3 REQUIRED)
catkin_package()
include_directories(${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
add_executable(lidar_depth_node src/lidar_depth_node.cpp)
target_compile_features(lidar_depth_node PRIVATE cxx_std_14)
target_link_libraries(lidar_depth_node ${catkin_LIBRARIES} ${OpenCV_LIBRARIES})
add_dependencies(lidar_depth_node ${catkin_EXPORTED_TARGETS})
install(TARGETS lidar_depth_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
catkin_install_python(PROGRAMS scripts/base_odometry.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
